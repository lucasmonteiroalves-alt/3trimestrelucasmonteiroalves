const ctx = canvas.getContext('2d'); const size = 8; const tileSize = canvas.width / size; // Estados do Jogo let board = []; let turn = 1; // 1 = Vermelhas, 2 = Pretas let selectedPiece = null; let validMoves = []; function initGame() { board = Array(size).fill(null).map(() => Array(size).fill(0)); turn = 1; selectedPiece = null; validMoves = []; // Preencher peças pretas (linhas 0, 1, 2) for (let r = 0; r < 3; r++) { for (let c = 0; c < size; c++) { if ((r + c) % 2 === 1) board[r][c] = 2; } } // Preencher peças vermelhas (linhas 5, 6, 7) for (let r = 5; r < size; r++) { for (let c = 0; c < size; c++) { if ((r + c) % 2 === 1) board[r][c] = 1; } } updateStatus(); draw(); } function updateStatus() { statusText.innerText = turn === 1 ? 'Turno: Vermelhas (Você)' : 'Turno: Pretas'; statusText.style.color = turn === 1 ? '#e74c3c' : '#2c3e50'; } function draw() { // Desenhar Tabuleiro for (let r = 0; r < size; r++) { for (let c = 0; c < size; c++) { ctx.fillStyle = (r + c) % 2 === 0 ? '#f0d9b5' : '#b58863'; ctx.fillRect(c * tileSize, r * tileSize, tileSize, tileSize); // Destacar movimentos válidos if (validMoves.some(m => m.r === r && m.c === c)) { ctx.fillStyle = 'rgba(46, 204, 113, 0.6)'; ctx.fillRect(c * tileSize, r * tileSize, tileSize, tileSize); } // Desenhar Peças let piece = board[r][c]; if (piece !== 0) { let isKing = piece > 2; let isRed = piece === 1 || piece === 3; ctx.beginPath(); ctx.arc(c * tileSize + tileSize / 2, r * tileSize + tileSize / 2, tileSize * 0.38, 0, Math.PI * 2); ctx.fillStyle = isRed ? '#e74c3c' : '#2c3e50'; ctx.fill(); ctx.strokeStyle = '#fff'; ctx.lineWidth = 2; ctx.stroke(); // Detalhe de Dama (Rei) if (isKing) { ctx.beginPath(); ctx.arc(c * tileSize + tileSize / 2, r * tileSize + tileSize / 2, tileSize * 0.15, 0, Math.PI * 2); ctx.fillStyle = '#f1c40f'; ctx.fill(); } // Destacar peça selecionada if (selectedPiece && selectedPiece.r === r && selectedPiece.c === c) { ctx.strokeStyle = '#2ecc71'; ctx.lineWidth = 4; ctx.stroke(); } } } } } function getMoves(r, c) { let piece = board[r][c]; if (piece === 0) return []; let moves = []; let isKing = piece > 2; let directions = []; if (piece === 1 || isKing) directions.push({ dr: -1, dc: -1 }, { dr: -1, dc: 1 }); // Sobe if (piece === 2 || isKing) directions.push({ dr: 1, dc: -1 }, { dr: 1, dc: 1 }); // Desce directions.forEach(d => { let nr = r + d.dr; let nc = c + d.dc; // Movimento simples if (nr >= 0 && nr < size && nc >= 0 && nc < size && board[nr][nc] === 0) { moves.push({ r: nr, c: nc, jump: false }); } // Movimento de captura (pulo) let jr = r + d.dr * 2; let jc = c + d.dc * 2; if (jr >= 0 && jr < size && jc >= 0 && jc < size && board[jr][jc] === 0) { let enemy = board[nr][nc]; if (enemy !== 0 && Math.floor((enemy - 1) % 2) !== Math.floor((piece - 1) % 2)) { moves.push({ r: jr, c: jc, jump: true, kr: nr, kc: nc }); } } }); return moves; } canvas.addEventListener('click', (e) => { const rect = canvas.getBoundingClientRect(); const x = e.clientX - rect.left; const y = e.clientY - rect.top; const c = Math.floor(x / tileSize); const r = Math.floor(y / tileSize); let clickedPiece = board[r][c]; let isCurrentTurnPiece = clickedPiece !== 0 && Math.floor((clickedPiece - 1) % 2) === (turn - 1); if (isCurrentTurnPiece) { selectedPiece = { r, c }; validMoves = getMoves(r, c); } else if (selectedPiece) { let move = validMoves.find(m => m.r === r && m.c === c); if (move) { // Executa o movimento let pValue = board[selectedPiece.r][selectedPiece.c]; board[r][c] = pValue; board[selectedPiece.r][selectedPiece.c] = 0; // Se for captura, remove a peça comida if (move.jump) { board[move.kr][move.kc] = 0; } // Promoção para Dama if (turn === 1 && r === 0) board[r][c] = 3; if (turn === 2 && r === 7) board[r][c] = 4; // Alterna o turno turn = turn === 1 ? 2 : 1; selectedPiece = null; validMoves = []; updateStatus(); // Simula IA simples para jogar contra se for o turno das Pretas if (turn === 2) { setTimeout(makeCPUMove, 500); } } else { selectedPiece = null; validMoves = []; } } draw(); }); function makeCPUMove() { let allMoves = []; for (let r = 0; r < size; r++) { for (let c = 0; c < size; c++) { if (board[r][c] === 2 || board[r][c] === 4) { let moves = getMoves(r, c); moves.forEach(m => allMoves.push({ from: { r, c }, to: m })); } } } if (allMoves.length > 0) { // Prioriza capturas let jumps = allMoves.filter(m => m.to.jump); let chosen = jumps.length > 0 ? jumps[Math.floor(Math.random() * jumps.length)] : allMoves[Math.floor(Math.random() * allMoves.length)]; let pValue = board[chosen.from.r][chosen.from.c]; board[chosen.to.r][chosen.to.c] = pValue; board[chosen.from.r][chosen.from.c] = 0; if (chosen.to.jump) { board[chosen.to.kr][chosen.to.kc] = 0; } if (chosen.to.r === 7) board[chosen.to.r][chosen.to.c] = 4; } turn = 1; updateStatus(); draw(); } // Inicializa o jogo ao carregar initGame();
